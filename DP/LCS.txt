BottomUp:
class Solution {
public:
    int longestCommonSubsequence(string text1, string text2) {
        int m=text1.size();
        int n=text2.size();
        vector<int>prev(n+1,0),curr(n+1,0);
        for(int i=m-1;i>=0;i--){
            for(int j=n-1;j>=0;j--){
                if(text1[i]==text2[j]){
                    curr[j]=1+prev[j+1];
                }else{
                    curr[j]=max(prev[j],curr[j+1]);
                }
            }
            prev=curr;
        }
        return prev[0];
    }
};

memo:

class Solution {
public:
    int t[1001][1001];
    int solve(string text1, string text2,int i,int j){
        if(i>=text1.size() || j>=text2.size()){ // either of text1 or text2 has been exhausted
            return 0;
        }
        if(t[i][j]!=-1) return t[i][j];
        if(text1[i]==text2[j]){
            return 1+solve(text1,text2,i+1,j+1);
        }
        return max(solve(text1,text2,i+1,j),solve(text1,text2,i,j+1));
    }
    int longestCommonSubsequence(string text1, string text2) {
        memset(t,-1,sizeof(t));
        return solve(text1,text2,0,0);
    }
};
