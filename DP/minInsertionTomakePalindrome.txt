class Solution {
public:
    int minInsertions(string s) {
        int n=s.size();
        vector<vector<int>>dp(n,vector<int>(n,0));// all 1 length substrings are marked with 0 
        for(int i=n-2;i>=0;i--){  // to consider substrings of length 2
            for(int j=i+1;j<n;j++){
                if(s[i]==s[j]) dp[i][j]=dp[i+1][j-1];
                else dp[i][j]=1+min(dp[i+1][j],dp[i][j-1]);
            }
        }
        return dp[0][n-1];
    }
};

memo:
class Solution {
public:
    int solve(string s,int i,int j,vector<vector<int>>& dp){
        if(i>=j) return 0;
        if(dp[i][j]!=-1) return dp[i][j];
        if(s[i]==s[j]) return dp[i][j]= solve(s,i+1,j-1,dp);
        return dp[i][j]= 1+min(solve(s,i+1,j,dp),solve(s,i,j-1,dp));
    }
    int minInsertions(string s) {
        int n=s.size();
        vector<vector<int>>dp(n+1,vector<int>(n+1,-1));
        return solve(s,0,n-1,dp);
    }
};
