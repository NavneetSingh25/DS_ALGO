class Solution {
public:
    int solve(vector<vector<int>>& obstacleGrid,int i,int j,int m,int n,vector<vector<int>> &t){
        if(i==m-1 && j==n-1){
            if(obstacleGrid[i][j]!=1){
                return 1;
            }else{
                return 0;
            }
        }
        if(i<0 || i>=m || j<0||j>=n) return 0;
        if(obstacleGrid[i][j]==1) return 0;
        if(t[i][j]!=-1) return t[i][j];
        int right=solve(obstacleGrid,i,j+1,m,n,t);
        int down=solve(obstacleGrid,i+1,j,m,n,t);
        return t[i][j]=right+down;
    }
    int uniquePathsWithObstacles(vector<vector<int>>& obstacleGrid) {
        int m=obstacleGrid.size();
        int n=obstacleGrid[0].size();
        vector<vector<int>>t(m+1,vector<int>(n+1,-1));
        return solve(obstacleGrid,0,0,m,n,t);
    }
};

class Solution {
public:
    int uniquePathsWithObstacles(vector<vector<int>>& obstacleGrid) {
        int m=obstacleGrid.size();
        int n=obstacleGrid[0].size();
        vector<vector<int>>t(m,vector<int>(n,-1));
        t[0][0]=obstacleGrid[0][0]==0?1:0;
        for(int col=1;col<n;col++){
            if(obstacleGrid[0][col]==0 && t[0][col-1]==1){
                t[0][col]=1;
            }else{
                t[0][col]=0;
            }
        }
        for(int row=1;row<m;row++){
            if(obstacleGrid[row][0]==0 && t[row-1][0]==1){
                t[row][0]=1;
            }else{
                t[row][0]=0;
            }
        }

        for(int i=1;i<m;i++){
            for(int j=1;j<n;j++){
                if(obstacleGrid[i][j]==0){
                    t[i][j]=t[i-1][j]+t[i][j-1];
                }else{
                    t[i][j]=0;
                }
            }
        }
        return t[m-1][n-1];
    }
};
