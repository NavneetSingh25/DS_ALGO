class Solution {
public:
    int solve(vector<vector<int>>& matrix,int i,int j,int n,vector<vector<int>>& dp){
        if(i==n-1){
            return matrix[n-1][j];
        }
        int sum=matrix[i][j];
        int minSum=INT_MAX;
        if(dp[i][j]!=-1) return dp[i][j];
        if(i+1<n && j-1>=0) minSum=min(minSum,sum+solve(matrix,i+1,j-1,n,dp));
        if(i+1<n ) minSum=min(minSum,sum+solve(matrix,i+1,j,n,dp));
        if(i+1<n && j+1<n) minSum=min(minSum,sum+solve(matrix,i+1,j+1,n,dp));
        return dp[i][j]=minSum;
    }
    int minFallingPathSum(vector<vector<int>>& matrix) {
       int n=matrix.size();
       vector<vector<int>>dp(n,vector<int>(n,-1));
       int result=INT_MAX;
       for(int j=0;j<n;j++){
        result=min(result,solve(matrix,0,j,n,dp));
       } 
       return result;
    }
};

BottomUp:

class Solution {
public:
    int minFallingPathSum(vector<vector<int>>& matrix) {
        int n=matrix.size();
        vector<vector<int>>dp(n,vector<int>(n,0));
        for(int j=0;j<n;j++){
            dp[0][j]=matrix[0][j];
        }
    
        for(int i=1;i<n;i++){
            for(int j=0;j<n;j++){
                 int a=INT_MAX;
                 int b=INT_MAX;

            if(i-1>=0 && j-1>=0) a=dp[i-1][j-1];
            if(i-1>=0 && j+1<n) b=dp[i-1][j+1];
            dp[i][j]=matrix[i][j]+min({dp[i-1][j],a,b});  //using {} for more than two elemnts to compare
            }
        }
        int result=INT_MAX;
        for(int j=0;j<n;j++){
            result=min(result,dp[n-1][j]);
        }

        return result;
    }
};

Optimized Space:

class Solution {
public:
    int minFallingPathSum(vector<vector<int>>& matrix) {
        int n=matrix.size();
        vector<int>prev(n,0);
        for(int j=0;j<n;j++){
            prev[j]=matrix[0][j];
        }
    
        for(int i=1;i<n;i++){
            vector<int>curr(n,0);
            for(int j=0;j<n;j++){
                
                 int a=INT_MAX;
                 int b=INT_MAX;

                if(i-1>=0 && j-1>=0) a=prev[j-1];
                if(i-1>=0 && j+1<n) b=prev[j+1];
                curr[j]=matrix[i][j]+min({prev[j],a,b});
            }
            prev=curr;
        }
        int result=INT_MAX;
        for(int j=0;j<n;j++){
            result=min(result,prev[j]);
        }

        return result;
    }
};
