class Solution {
public:
    bool solve(vector<int>& nums,int x,int i,vector<vector<int>>& t){
        if(x==0) return true;
        if(i>=nums.size()) return false;
        if(t[i][x]!=-1) return t[i][x];
        bool take=false;
        if(x>=nums[i]){
             take=solve(nums,x-nums[i],i+1,t);
        }
        bool not_take=solve(nums,x,i+1,t);
        return t[i][x]=take||not_take;
    }
    bool canPartition(vector<int>& nums) {
        int n=nums.size();
        
        int s=accumulate(begin(nums),end(nums),0);
        if(s%2!=0) return false;
        int x=s/2;
        vector<vector<int>>t(n+1,vector<int>(x+1,-1));
        return solve(nums,x,0,t);
    }
};

BottomUp:
class Solution {
public:
    bool canPartition(vector<int>& nums) {
        int n = nums.size();
        int sum = accumulate(nums.begin(), nums.end(), 0);

        if (sum % 2 != 0) return false;
        int x = sum / 2;

        vector<vector<bool>> dp(n + 1, vector<bool>(x + 1, false));

        // Base Case: With 0 elements, you can always make sum 0
        for (int i = 0; i <= n; i++) dp[i][0] = true;

        // Fill table bottom-up
        for (int i = n - 1; i >= 0; i--) {
            for (int target = 0; target <= x; target++) {
                bool not_take = dp[i + 1][target];
                bool take = (target >= nums[i]) ? dp[i + 1][target - nums[i]] : false;
                dp[i][target] = take || not_take;
            }
        }

        return dp[0][x];
    }
};
